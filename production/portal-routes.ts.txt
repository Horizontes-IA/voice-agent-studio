const identityCookie='customer_session';
async function customer(c:any){
  const token=getCookie(c,identityCookie);
  if(!token)return null;
  return c.env.DB.prepare('SELECT p.* FROM customer_sessions s JOIN patients p ON p.id=s.patient_id WHERE s.token_hash=? AND s.expires_at>? AND p.is_demo=0').bind(await digest(token),now()).first() as Promise<Patient|null>;
}
app.post('/api/portal/start',async c=>{
  if(c.env.APP_ENV!=='production'||c.env.BUSINESS_APPROVED!=='true')return c.json({error:'PORTAL_NOT_ENABLED'},503);
  const p=z.object({phone:z.string().regex(/^\+[1-9][0-9]{7,14}$/),consent:z.literal(true)}).parse(await c.req.json());
  const ip=c.req.header('CF-Connecting-IP')??'local';
  if(!await rateLimit(c.env.DB,'otp-ip:'+ip,5,3600)||!await rateLimit(c.env.DB,'otp-phone:'+await digest(p.phone),3,3600))return c.json({error:'Intenta más tarde o contacta al negocio.'},429);
  if(!await verifyPhone(c.env,p.phone))return c.json({error:'No se pudo iniciar la verificación.'},502);
  const id=crypto.randomUUID();
  await c.env.DB.prepare('INSERT INTO otp_challenges(id,phone,expires_at) VALUES(?,?,?)').bind(id,p.phone,new Date(Date.now()+600000).toISOString()).run();
  return c.json({challenge_id:id,message:'Se solicitó un código. Su entrega depende del proveedor.'});
});
app.post('/api/portal/check',async c=>{
  if(c.env.APP_ENV!=='production'||c.env.BUSINESS_APPROVED!=='true')return c.json({error:'PORTAL_NOT_ENABLED'},503);
  const p=z.object({challenge_id:z.string().uuid(),code:z.string().regex(/^[0-9]{4,10}$/),name:z.string().min(3).max(100)}).parse(await c.req.json());
  if(!await rateLimit(c.env.DB,'otp-check:'+(c.req.header('CF-Connecting-IP')??'local'),20,3600))return c.json({error:'Intenta más tarde.'},429);
  const challenge=await c.env.DB.prepare('UPDATE otp_challenges SET attempts=attempts+1 WHERE id=? AND consumed=0 AND attempts<5 AND expires_at>? RETURNING phone').bind(p.challenge_id,now()).first<{phone:string}>();
  if(!challenge||!await verifyPhone(c.env,challenge.phone,p.code))return c.json({error:'Código no aprobado o caducado.'},403);
  const consumed=await c.env.DB.prepare('UPDATE otp_challenges SET consumed=1 WHERE id=? AND consumed=0').bind(p.challenge_id).run();
  if(!consumed.meta.changes)return c.json({error:'CHALLENGE_ALREADY_USED'},409);
  const id=crypto.randomUUID(),initials=p.name.trim().split(/\s+/).map(n=>n[0]).slice(0,2).join('');
  await c.env.DB.prepare('INSERT INTO patients(id,name,phone,email,initials,is_demo) VALUES(?,?,?,NULL,?,0) ON CONFLICT(phone) WHERE is_demo=0 DO NOTHING').bind(id,p.name.trim(),challenge.phone,initials).run();
  const owner=await c.env.DB.prepare('SELECT id FROM patients WHERE phone=? AND is_demo=0').bind(challenge.phone).first<{id:string}>();
  if(!owner)throw new BusinessError('CUSTOMER_NOT_FOUND',500);
  const token=crypto.randomUUID()+crypto.randomUUID();
  await c.env.DB.prepare('INSERT INTO customer_sessions(token_hash,patient_id,expires_at) VALUES(?,?,?)').bind(await digest(token),owner.id,new Date(Date.now()+1800000).toISOString()).run();
  setCookie(c,identityCookie,token,{httpOnly:true,secure:true,sameSite:'Strict',path:'/api/portal',maxAge:1800});
  return c.json({verified:true});
});
app.post('/api/portal/logout',async c=>{
  const token=getCookie(c,identityCookie);
  if(token)await c.env.DB.prepare('DELETE FROM customer_sessions WHERE token_hash=?').bind(await digest(token)).run();
  deleteCookie(c,identityCookie,{path:'/api/portal'});return c.json({ok:true});
});
app.post('/api/portal/voice',async c=>{
  if(c.env.APP_ENV!=='production'||c.env.BUSINESS_APPROVED!=='true')return c.json({error:'PORTAL_NOT_ENABLED'},503);
  const owner=await customer(c);if(!owner)return c.json({error:'Verifica tu teléfono para continuar.'},401);
  if(!await rateLimit(c.env.DB,'voice-owner:'+owner.id,8,3600))return c.json({error:'Límite de conversaciones alcanzado.'},429);
  const active=await c.env.DB.prepare("SELECT id FROM voice_sessions WHERE patient_id=? AND expires_at>? AND mode='verified'").bind(owner.id,now()).first();
  if(active)return c.json({error:'Ya existe una conversación activa. Espera a que termine.'},409);
  const [settings,key]=await Promise.all([getSettings(c.env.DB),apiKey(c.env)]);
  if(!settings.active||!settings.remote_agent_id||!key)return c.json({error:'El asistente no está disponible.'},503);
  const id=crypto.randomUUID(),token=crypto.randomUUID()+crypto.randomUUID();
  const signed=await eleven<{signed_url:string}>(key,'/v1/convai/conversation/get-signed-url?agent_id='+encodeURIComponent(settings.remote_agent_id));
  await c.env.DB.prepare("INSERT INTO voice_sessions(id,token_hash,patient_id,verified,mode,expires_at) VALUES(?,?,?,1,'verified',?)").bind(id,await digest(token),owner.id,new Date(Date.now()+Math.min(settings.max_duration+120,1500)*1000).toISOString()).run();
  return c.json({signed_url:signed.signed_url,session_id:id,dynamic_variables:{session_token:token,patient_name:owner.name,clinic_date:today(),pending_operation_id:''}});
});
app.post('/api/portal/voice/:id/attach',async c=>{
  const owner=await customer(c);if(!owner)return c.json({error:'UNAUTHORIZED'},401);
  const session=await c.env.DB.prepare('SELECT id FROM voice_sessions WHERE id=? AND patient_id=? AND expires_at>?').bind(c.req.param('id'),owner.id,now()).first();
  if(!session)return c.json({error:'SESSION_NOT_FOUND'},404);
  const p=z.object({conversation_id:z.string().min(1).max(100)}).parse(await c.req.json());
  const key=await apiKey(c.env),settings=await getSettings(c.env.DB);if(!key)throw new BusinessError('VOICE_NOT_CONFIGURED',503);
  const remote=await eleven<{agent_id:string;conversation_initiation_client_data?:{dynamic_variables?:{session_token?:string}}}>(key,'/v1/convai/conversations/'+encodeURIComponent(p.conversation_id));
  const remoteToken=remote.conversation_initiation_client_data?.dynamic_variables?.session_token;
  const expected=await c.env.DB.prepare('SELECT token_hash FROM voice_sessions WHERE id=?').bind(c.req.param('id')).first<{token_hash:string}>();
  if(remote.agent_id!==settings.remote_agent_id||!remoteToken||await digest(remoteToken)!==expected?.token_hash)return c.json({error:'CONVERSATION_NOT_AUTHORIZED'},403);
  await c.env.DB.batch([c.env.DB.prepare('UPDATE voice_sessions SET conversation_id=? WHERE id=?').bind(p.conversation_id,c.req.param('id')),c.env.DB.prepare('INSERT INTO conversations(id,patient_id,is_demo) VALUES(?,?,0) ON CONFLICT(id) DO NOTHING').bind(p.conversation_id,owner.id)]);
  return c.json({ok:true});
});
app.post('/api/portal/voice/:id/end',async c=>{
  const owner=await customer(c);if(!owner)return c.json({error:'UNAUTHORIZED'},401);
  await c.env.DB.prepare('UPDATE voice_sessions SET expires_at=? WHERE id=? AND patient_id=?').bind(now(),c.req.param('id'),owner.id).run();return c.json({ok:true});
});
